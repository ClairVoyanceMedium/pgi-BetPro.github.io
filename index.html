📖 Installation
Mise en place en 15 minutes.
1 — Config Worker
const WORKER_URL = 'https://pgi-betpro.peritoregroupindustries.workers.dev';
const AUTH_TOKEN = 'PgiBetProWinMoney';
2 — Déployer
npm install -g wrangler && wrangler login
wrangler kv:namespace create BETTING_KV
wrangler secret put ODDS_API_KEY
wrangler secret put API_FOOTBALL_KEY
wrangler secret put AI_PROVIDER   # groq | gemini | anthropic
wrangler secret put AI_API_KEY
wrangler secret put RESEND_API_KEY
wrangler secret put EMAIL_TO
wrangler deploy
